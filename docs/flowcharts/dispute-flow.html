Dispute Resolution State Machine

Three dispute modes: standard, mediator, jury. Each has different on-chain covenant paths and indexer states.

stateDiagram-v2
  state "Escrow Active" as ACTIVE
  state "Disputed" as DISPUTED

  state "Standard Mode" as STD {
    [*] --> STD_WAIT : buyer or seller disputes
    STD_WAIT --> STD_SETTLE : parties co-sign release()
    STD_WAIT --> STD_REFUND : timeout reached → buyer calls refund()
    STD_SETTLE : Settled ✅
    STD_REFUND : Refunded ↩️
  }

  state "Mediator Mode" as MED {
    [*] --> MED_EVIDENCE : dispute raised
    MED_EVIDENCE --> MED_LOG : parties + mediator exchange evidence
    MED_LOG --> MED_SELLER : mediator signs disputeSellerWins
    MED_LOG --> MED_BUYER : mediator signs disputeBuyerWins
    MED_LOG --> MED_TIMEOUT : timeout → refundAfterTimeout
    MED_SELLER : Seller paid (minus fees)
    MED_BUYER : Buyer refunded (minus fees)
    MED_TIMEOUT : Mediator + buyer co-sign refund
  }

  state "Jury Mode" as JURY {
    [*] --> SELECTING : top N*2 jurors → random N
    SELECTING --> VOTING : case opened (3/5/9 jurors)
    VOTING --> VERDICT_SELLER : threshold reached
    VOTING --> VERDICT_BUYER : threshold reached
    VOTING --> VERDICT_SELLER : 72h timeout → seller_wins default
    VERDICT_SELLER : Verdict: seller_wins
    VERDICT_BUYER : Verdict: buyer_wins
  }

  ACTIVE --> DISPUTED : dispute raised
  DISPUTED --> STD : mode=standard
  DISPUTED --> MED : mode=mediator / mediator_key set
  DISPUTED --> JURY : mode=jury

  note right of ACTIVE
    Escrow can be in PendingConfirmation
    or Active status when dispute is raised
  end note

  note right of JURY
    72h timeout defaults to seller_wins.
    Treasury executes on-chain verdict
    via disputeSellerWins/disputeBuyerWins.
  end note
